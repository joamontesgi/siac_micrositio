\section{Frontend (área de contenido)}
\begin{itemize}
  \item \textbf{Blade} --- plantillas Laravel.
  \item \textbf{Bootstrap 4} --- grid, formularios y componentes.
  \item \textbf{Plantilla UNAL} --- cabecera, pie y accesibilidad (\texttt{public/unal/}).
  \item \textbf{CSS SIAC} --- paleta institucional (\texttt{\#232a5e}, \texttt{\#00b8a9}, \texttt{\#ffc107}, \texttt{\#a8b0d6}).
  \item \textbf{JavaScript} --- filtros de indicadores (\texttt{siac-indicators.js}), jQuery/Bootstrap UNAL.
\end{itemize}
